export function MiniGlobe ({ size = 150 }) {
  const canvasRef = useRef()
  const { t } = useI18n()
  useEffect(() => {
    let phi = 0
    let width = 0
    const onResize = () => canvasRef.current && (width = canvasRef.current.offsetWidth)
    window.addEventListener('resize', onResize)
    onResize()
    const globe = createGlobe(canvasRef.current, {
      devicePixelRatio: 1.5,
      width: width * 2,
      height: width * 2,
      phi: 0,
      theta: 0.3,
      dark: 1,
      diffuse: 3,
      mapSamples: 12000,
      mapBrightness: 1.2,
      baseColor: [1, 1, 1],
      markerColor: [251 / 255, 100 / 255, 21 / 255],
      glowColor: [1.2, 1.2, 1.2],
      markers: [{ location: [39.9042, 116.4074], size: 0.05 }],
      onRender: (state) => {
        phi += 0.003
        state.phi = phi
        state.width = width * 2
        state.height = width * 2
      }
    })
    setTimeout(() => { if (canvasRef.current) canvasRef.current.style.opacity = '1' })
    return () => { globe.destroy(); window.removeEventListener('resize', onResize) }
  }, [])
  return (
    <div className="flex flex-col items-center gap-2 h-full justify-center">
      <div
        className="rounded-2xl bg-warm-900 dark:bg-warm-950 overflow-hidden p-2"
        style={{ width: size, height: size }}
      >
        <canvas
          ref={canvasRef}
          style={{ width: '100%', height: '100%', opacity: 0, transition: 'opacity 0.8s ease', cursor: 'default' }}
        />
      </div>
      <span className="text-xs text-warm-500 dark:text-warm-400">
        {t('about_location')}
      </span>
    </div>
  )
}
